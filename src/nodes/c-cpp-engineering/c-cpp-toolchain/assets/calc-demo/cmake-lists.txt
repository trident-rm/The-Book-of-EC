cmake_minimum_required(VERSION 3.28)
project(calc_demo LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(CALC_TRACE "Enable calculation logs" OFF)

add_library(calc STATIC src/calc.c)
target_include_directories(calc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(calc PRIVATE -Wall -Wextra -Wpedantic)
if(CALC_TRACE)
    target_compile_definitions(calc PRIVATE CALC_TRACE=1)
endif()

add_executable(calc-demo src/main.c)
target_compile_options(calc-demo PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(calc-demo PRIVATE calc)
